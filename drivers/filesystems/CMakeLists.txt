function(add_optional_filesystem name option)
    if(${option})
        add_subdirectory(${name})
    endif()
endfunction()

add_optional_filesystem(btrfs    ENABLE_FS_BTRFS)
add_optional_filesystem(cdfs     ENABLE_FS_CDFS)
add_optional_filesystem(exfat    ENABLE_FS_EXFAT)
add_optional_filesystem(ext2     ENABLE_FS_EXT2)
add_optional_filesystem(fastfat  ENABLE_FS_FASTFAT)
add_optional_filesystem(fs_rec   ENABLE_FS_FS_REC)
add_optional_filesystem(msfs     ENABLE_FS_MSFS)
add_optional_filesystem(mup      ENABLE_FS_MUP)
add_optional_filesystem(nfs      ENABLE_FS_NFS)
add_optional_filesystem(npfs     ENABLE_FS_NPFS)
add_optional_filesystem(ntfs     ENABLE_FS_NTFS)
add_optional_filesystem(ntfs_new ENABLE_FS_NTFS_NEW)
add_optional_filesystem(udfs     ENABLE_FS_UDFS)
add_optional_filesystem(vfatfs   ENABLE_FS_VFATFS)
